### 一、Jest 测试框架

```javascript
// math.test.js
const { add, multiply } = require('./math');

describe('Math functions', () => {
    test('adds 1 + 2 to equal 3', () => {
        expect(add(1, 2)).toBe(3);
    });
    
    test('multiplies 2 * 3 to equal 6', () => {
        expect(multiply(2, 3)).toBe(6);
    });
    
    test('throws error on invalid input', () => {
        expect(() => add('a', 1)).toThrow('Invalid input');
    });
});

// 异步测试
describe('Async operations', () => {
    test('fetches user data', async () => {
        const user = await fetchUser(1);
        expect(user).toEqual(expect.objectContaining({
            id: 1,
            username: expect.any(String)
        }));
    });
    
    test('fetch resolves with data', () => {
        return expect(fetchData()).resolves.toHaveLength(10);
    });
});

// 生命周期
beforeAll(() => { /* 全局初始化 */ });
afterAll(() => { /* 全局清理 */ });
beforeEach(() => { /* 每个测试前 */ });
afterEach(() => { /* 每个测试后 */ });
```

- **常用断言：**
  - `toBe(value)`：严格相等（===）
  - `toEqual(object)`：深度相等
  - `toBeTruthy()` / `toBeFalsy()`
  - `toContain(item)`：数组/字符串包含
  - `toThrow(error)`：抛出异常
  - `toHaveBeenCalled()`：函数被调用
  - `toHaveBeenCalledWith(args)`：指定参数调用
  - `expect.any(Type)`：任意类型

### 二、Supertest API 测试

```javascript
const request = require('supertest');
const app = require('../app');

describe('User API', () => {
    let token;
    
    beforeAll(async () => {
        // 获取认证 token
        const res = await request(app)
            .post('/api/login')
            .send({ email: 'test@example.com', password: 'password123' });
        token = res.body.token;
    });
    
    test('GET /api/users - 获取用户列表', async () => {
        const res = await request(app)
            .get('/api/users')
            .set('Authorization', `Bearer ${token}`)
            .query({ page: 1, size: 10 });
        
        expect(res.status).toBe(200);
        expect(res.body.data).toBeInstanceOf(Array);
        expect(res.body.data.length).toBeLessThanOrEqual(10);
    });
    
    test('POST /api/users - 创建用户', async () => {
        const res = await request(app)
            .post('/api/users')
            .set('Authorization', `Bearer ${token}`)
            .send({ username: 'newuser', email: 'new@example.com', password: 'Pass123!' });
        
        expect(res.status).toBe(201);
        expect(res.body.username).toBe('newuser');
        expect(res.body).not.toHaveProperty('password');
    });
    
    test('GET /api/users/999 - 用户不存在', async () => {
        const res = await request(app)
            .get('/api/users/999')
            .set('Authorization', `Bearer ${token}`);
        
        expect(res.status).toBe(404);
        expect(res.body.error).toMatch(/不存在/);
    });
    
    test('未认证请求返回401', async () => {
        const res = await request(app).get('/api/users');
        expect(res.status).toBe(401);
    });
});
```

### 三、Mock 与 Stub

```javascript
// 模块 Mock
jest.mock('../services/emailService');
const emailService = require('../services/emailService');

test('sends welcome email on registration', async () => {
    emailService.sendWelcome.mockResolvedValue(true);
    
    await request(app).post('/api/register').send({ email: 'test@example.com' });
    
    expect(emailService.sendWelcome).toHaveBeenCalledWith('test@example.com');
});

// 函数 Mock
const mockFn = jest.fn();
mockFn.mockReturnValue(42);
mockFn.mockResolvedValue({ id: 1 }); // 异步
mockFn.mockImplementation((x) => x * 2);

// 数据库 Mock
jest.mock('../models/User');
const User = require('../models/User');

User.find.mockResolvedValue([{ id: 1, username: 'test' }]);
User.findById.mockImplementation((id) => 
    Promise.resolve({ id, username: 'test' })
);

// Spy
const spy = jest.spyOn(console, 'log');
expect(spy).toHaveBeenCalledWith('expected message');
spy.mockRestore();
```

### 四、数据库测试隔离

```javascript
// 使用 mongodb-memory-server
const { MongoMemoryServer } = require('mongodb-memory-server');
const mongoose = require('mongoose');

let mongoServer;

beforeAll(async () => {
    mongoServer = await MongoMemoryServer.create();
    await mongoose.connect(mongoServer.getUri());
});

afterAll(async () => {
    await mongoose.disconnect();
    await mongoServer.stop();
});

afterEach(async () => {
    // 清理所有集合
    const collections = mongoose.connection.collections;
    for (const key in collections) {
        await collections[key].deleteMany({});
    }
});

// 使用 Docker Testcontainers
const { GenericContainer } = require('testcontainers');

let mongoContainer;

beforeAll(async () => {
    mongoContainer = await new GenericContainer('mongo:7')
        .withExposedPorts(27017)
        .start();
    const uri = `mongodb://localhost:${mongoContainer.getMappedPort(27017)}/testdb`;
    await mongoose.connect(uri);
}, 60000);
```

```javascript
    testEnvironment: 'node',
    coverageDirectory: 'coverage',
    coverageThreshold: {
        global: {
            branches: 70,
            functions: 80,
            lines: 80,
            statements: 80
        }
    },
    testMatch: ['**/__tests__/**/*.test.js', '**/*.spec.js'],
    collectCoverageFrom: [
        'src/**/*.js',
        '!src/**/*.test.js',
        '!src/config/**'
    ]
};
```
