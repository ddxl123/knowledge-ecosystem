### 一、WebSocket 基础

```javascript
const WebSocket = require('ws');
const http = require('http');

const server = http.createServer();
const wss = new WebSocket.Server({ server });

wss.on('connection', (ws, req) => {
    console.log('Client connected');
    
    ws.on('message', (data) => {
        const message = JSON.parse(data);
        console.log('Received:', message);
        
        // 广播给所有客户端
        wss.clients.forEach(client => {
            if (client.readyState === WebSocket.OPEN) {
                client.send(JSON.stringify({
                    type: 'message',
                    data: message,
                    timestamp: Date.now()
                }));
            }
        });
    });
    
    ws.on('close', () => console.log('Client disconnected'));
    ws.on('error', (err) => console.error('WebSocket error:', err));
    
    // 心跳检测
    ws.isAlive = true;
    ws.on('pong', () => { ws.isAlive = true; });
});

// 心跳定时器
setInterval(() => {
    wss.clients.forEach(ws => {
        if (!ws.isAlive) return ws.terminate();
        ws.isAlive = false;
        ws.ping();
    });
}, 30000);

server.listen(8080);
```

### 二、Socket.IO

```javascript
const { Server } = require('socket.io');
const io = new Server(server, {
    cors: { origin: 'http://localhost:3000' }
});

// 认证中间件
io.use((socket, next) => {
    const token = socket.handshake.auth.token;
    try {
        const user = jwt.verify(token, JWT_SECRET);
        socket.user = user;
        next();
    } catch {
        next(new Error('认证失败'));
    }
});

io.on('connection', (socket) => {
    console.log(`User ${socket.user.username} connected`);
    
    // 加入房间
    socket.join('general');
    
    // 聊天消息
    socket.on('chat:message', async (data) => {
        const message = await Message.create({
            content: data.content,
            author: socket.user.id,
            room: data.room
        });
        io.to(data.room).emit('chat:message', {
            ...message.toObject(),
            authorName: socket.user.username
        });
    });
    
    // 私聊
    socket.on('private:message', ({ to, content }) => {
        socket.to(to).emit('private:message', {
            from: socket.user.id,
            fromName: socket.user.username,
            content,
            timestamp: Date.now()
        });
    });
    
    // 输入状态
    socket.on('typing:start', (room) => {
        socket.to(room).emit('typing:start', { userId: socket.user.id });
    });
    
    socket.on('typing:stop', (room) => {
        socket.to(room).emit('typing:stop', { userId: socket.user.id });
    });
    
    // 加入/离开房间
    socket.on('room:join', (room) => {
        socket.join(room);
        io.to(room).emit('room:userJoined', { userId: socket.user.id });
    });
    
    socket.on('disconnect', () => {
        console.log(`User ${socket.user.username} disconnected`);
    });
});
```

### 三、Server-Sent Events (SSE)

```javascript
// SSE 适合单向推送（服务器→客户端）
app.get('/api/events', (req, res) => {
    res.writeHead(200, {
        'Content-Type': 'text/event-stream',
        'Cache-Control': 'no-cache',
        'Connection': 'keep-alive'
    });
    
    const sendEvent = (data) => {
        res.write(`data: ${JSON.stringify(data)}\n\n`);
    };
    
    // 发送初始连接事件
    sendEvent({ type: 'connected', message: '已连接' });
    
    // 订阅消息队列
    const subscriber = redis.duplicate();
    subscriber.subscribe('notifications');
    subscriber.on('message', (channel, message) => {
        sendEvent(JSON.parse(message));
    });
    
    // 心跳保持连接
    const heartbeat = setInterval(() => {
        res.write(':heartbeat\n\n');
    }, 30000);
    
    req.on('close', () => {
        clearInterval(heartbeat);
        subscriber.unsubscribe();
        subscriber.quit();
    });
});
```

### 四、Redis Pub/Sub 扩展

```javascript
const Redis = require('ioredis');
const pubClient = new Redis();
const subClient = pubClient.duplicate();

// Socket.IO 多实例扩展
const { createAdapter } = require('@socket.io/redis-adapter');
io.adapter(createAdapter(pubClient, subClient));

// 发布消息
async function broadcastNotification(userId, notification) {
    await pubClient.publish('notifications', JSON.stringify({
        targetUser: userId,
        ...notification
    }));
}
```

### 五、实时应用场景

- **在线聊天：** 房间管理、消息持久化、已读回执
- **实时通知：** 系统推送、@提醒、点赞通知
- **协同编辑：** OT/CRDT 算法、光标同步
- **实时数据看板：** 股票行情、监控数据、日志流
- **在线状态：** 用户在线/离线、最后活跃时间
- **游戏：** 实时对战、状态同步

- **性能考虑：**
  - WebSocket 连接数限制（单机约 10K-50K）
  - 使用 Redis Pub/Sub 实现跨实例广播
  - 消息压缩（permessage-deflate）
  - 连接认证与鉴权
  - 消息限流防刷
