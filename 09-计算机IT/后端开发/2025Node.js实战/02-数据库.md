# 数据库操作

## 核心知识点

### 一、MongoDB与Mongoose

#### 1. 连接与模型
```javascript
const mongoose = require('mongoose');

mongoose.connect('mongodb://localhost:27017/mydb');

const userSchema = new mongoose.Schema({
    name: { type: String, required: true },
    email: { type: String, required: true, unique: true },
    age: { type: Number, min: 0 },
    createdAt: { type: Date, default: Date.now }
});

userSchema.methods.toPublic = function() {
    return { id: this._id, name: this.name, email: this.email };
};

const User = mongoose.model('User', userSchema);
```

#### 2. CRUD操作
```javascript
// 创建
const user = await User.create({ name: 'Alice', email: 'alice@example.com' });

// 查询
const users = await User.find({ age: { $gte: 18 } }).sort('-createdAt').limit(10);
const user = await User.findById(id);
const user = await User.findOne({ email: 'alice@example.com' });

// 更新
await User.findByIdAndUpdate(id, { name: 'Bob' }, { new: true });
await User.updateMany({ age: { $lt: 18 } }, { status: 'minor' });

// 删除
await User.findByIdAndDelete(id);
await User.deleteMany({ status: 'inactive' });
```

### 二、MySQL与Sequelize

#### 1. 模型定义
```javascript
const { Sequelize, DataTypes } = require('sequelize');

const sequelize = new Sequelize('mydb', 'user', 'password', {
    host: 'localhost',
    dialect: 'mysql'
});

const User = sequelize.define('User', {
    name: { type: DataTypes.STRING, allowNull: false },
    email: { type: DataTypes.STRING, unique: true, allowNull: false },
    age: { type: DataTypes.INTEGER, defaultValue: 0 }
});

const Post = sequelize.define('Post', {
    title: DataTypes.STRING,
    content: DataTypes.TEXT
});

// 关联
User.hasMany(Post);
Post.belongsTo(User);
```

#### 2. 查询操作
```javascript
// 创建
const user = await User.create({ name: 'Alice', email: 'alice@example.com' });

// 查询
const users = await User.findAll({ where: { age: { [Op.gte]: 18 } } });
const user = await User.findByPk(id);
const user = await User.findOne({ where: { email } });

// 包含关联
const userWithPosts = await User.findByPk(id, {
    include: [{ model: Post, as: 'posts' }]
});

// 更新
await user.update({ name: 'Bob' });
await User.update({ status: 'active' }, { where: { age: { [Op.gte]: 18 } } });

// 删除
await user.destroy();
```

### 三、Prisma ORM

#### 1. Schema定义
```prisma
// schema.prisma
model User {
    id        Int      @id @default(autoincrement())
    name      String
    email     String   @unique
    posts     Post[]
    createdAt DateTime @default(now())
}

model Post {
    id        Int      @id @default(autoincrement())
    title     String
    content   String?
    author    User     @relation(fields: [authorId], references: [id])
    authorId  Int
    createdAt DateTime @default(now())
}
```

#### 2. 查询操作
```javascript
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

// 创建
const user = await prisma.user.create({
    data: { name: 'Alice', email: 'alice@example.com' }
});

// 查询
const users = await prisma.user.findMany({
    where: { age: { gte: 18 } },
    include: { posts: true },
    orderBy: { createdAt: 'desc' },
    take: 10
});

// 更新
const updated = await prisma.user.update({
    where: { id: 1 },
    data: { name: 'Bob' }
});

// 删除
await prisma.user.delete({ where: { id: 1 } });
```

### 四、Redis缓存

```javascript
const { createClient } = require('redis');

const redis = createClient();
await redis.connect();

// 基本操作
await redis.set('key', 'value', { EX: 3600 }); // 1小时过期
const value = await redis.get('key');

// 缓存中间件
const cacheMiddleware = (duration) => async (req, res, next) => {
    const key = req.originalUrl;
    const cached = await redis.get(key);

    if (cached) {
        return res.json(JSON.parse(cached));
    }

    res.sendResponse = res.json;
    res.json = (body) => {
        redis.setEx(key, duration, JSON.stringify(body));
        res.sendResponse(body);
    };
    next();
};

app.get('/api/users', cacheMiddleware(300), async (req, res) => {
    const users = await User.find();
    res.json(users);
});
```
