### 一、MongoDB 与 Mongoose

```javascript
const mongoose = require('mongoose');

// Schema 定义
const userSchema = new mongoose.Schema({
    username: { type: String, required: true, unique: true, trim: true },
    email: { type: String, required: true, unique: true, lowercase: true },
    password: { type: String, required: true, select: false },
    role: { type: String, enum: ['user', 'admin'], default: 'user' },
    profile: {
        avatar: String,
        bio: { type: String, maxlength: 500 }
    },
    tags: [{ type: mongoose.Schema.Types.ObjectId, ref: 'Tag' }]
}, { timestamps: true });

// 索引
userSchema.index({ email: 1 });
userSchema.index({ username: 'text', 'profile.bio': 'text' });

// 虚拟字段
userSchema.virtual('posts', {
    ref: 'Post',
    localField: '_id',
    foreignField: 'author'
});

// 中间件
userSchema.pre('save', async function(next) {
    if (this.isModified('password')) {
        this.password = await bcrypt.hash(this.password, 12);
    }
    next();
});

const User = mongoose.model('User', userSchema);

// CRUD 操作
const user = await User.create({ username: 'test', email: 'test@example.com', password: '123' });
const found = await User.findOne({ email: 'test@example.com' }).select('+password');
const updated = await User.findByIdAndUpdate(id, { $set: { 'profile.bio': 'new bio' } }, { new: true });
await User.deleteOne({ _id: id });

// 聚合查询
const stats = await User.aggregate([
    { $match: { role: 'user' } },
    { $group: { _id: '$status', count: { $sum: 1 } } },
    { $sort: { count: -1 } }
]);
```

### 二、Prisma ORM

```prisma

    id        Int      @id @default(autoincrement())

}

model Post {
    id        Int      @id @default(autoincrement())
    title     String
    content   String?
    published Boolean  @default(false)
    author    User     @relation(fields: [authorId], references: [id])
    authorId  Int
    tags      Tag[]
    createdAt DateTime @default(now())
}

model Tag {
    id    Int    @id @default(autoincrement())
    name  String @unique
    posts Post[]
}
```

### 三、Sequelize ORM

```javascript
const { Sequelize, DataTypes } = require('sequelize');
const sequelize = new Sequelize('mysql://user:pass@localhost:3306/mydb');

// 模型定义
const User = sequelize.define('User', {
    username: { type: DataTypes.STRING, allowNull: false, unique: true },
    email: { type: DataTypes.STRING, allowNull: false, validate: { isEmail: true } },
    role: { type: DataTypes.ENUM('user', 'admin'), defaultValue: 'user' }
}, { tableName: 'users', timestamps: true });

// 关联
User.hasMany(Post, { foreignKey: 'userId' });
Post.belongsTo(User, { foreignKey: 'userId' });
Post.belongsToMany(Tag, { through: 'PostTags' });

// 查询
const users = await User.findAll({
    where: { role: 'user' },
    include: [{ model: Post, where: { published: true }, required: false }],
    order: [['createdAt', 'DESC']],
    limit: 20,
    offset: 0
});

// 事务
const t = await sequelize.transaction();
try {
    const user = await User.create(userData, { transaction: t });
    await Profile.create({ userId: user.id, ...profileData }, { transaction: t });
    await t.commit();
} catch (err) {
    await t.rollback();
    throw err;
}
```

### 五、Redis 缓存集成

```javascript
const Redis = require('ioredis');
const redis = new Redis();

// 缓存模式
async function getUserById(id) {
    const cacheKey = `user:${id}`;
    
    // 先查缓存
    const cached = await redis.get(cacheKey);
    if (cached) return JSON.parse(cached);
    
    // 缓存未命中，查数据库
    const user = await User.findById(id).lean();
    if (user) {
        await redis.setex(cacheKey, 3600, JSON.stringify(user)); // 缓存1小时
    }
    return user;
}

// 缓存失效
async function updateUser(id, data) {
    const user = await User.findByIdAndUpdate(id, data, { new: true });
    await redis.del(`user:${id}`); // 删除缓存
    return user;
}
```
