### 一、JUnit 6 与测试增强

```java
// JUnit 6 新特性
class UserServiceTest {
    
    // @MockitoBean 替代 @MockBean（Spring Boot 3.4+）
    @MockitoBean
    private UserRepository userRepository;
    
    @Test
    @DisplayName("用户查询测试")
    void testFindUser() {
        when(userRepository.findById(1L))
            .thenReturn(Optional.of(new User(1L, "test", "test@example.com")));
        
        UserDTO result = userService.findById(1L);
        
        assertThat(result).isNotNull()
            .extracting(UserDTO::username)
            .isEqualTo("test");
    }
    
    // 参数化测试增强
    @ParameterizedTest
    @CsvSource(textBlock = """
        testuser, test@example.com, true
        , test@example.com, false
        testuser, invalid-email, false
        """)
    void validateUser(String username, String email, boolean expected) {
        assertThat(userService.isValid(username, email)).isEqualTo(expected);
    }
    
    // 嵌套测试组织
    @Nested
    class WhenUserExists {
        @BeforeEach
        void setup() {
            when(userRepository.existsById(1L)).thenReturn(true);
        }
        
        @Test
        void shouldReturnUser() { }
        
        @Test
        void shouldUpdateUser() { }
    }
}
```

### 二、原生镜像测试

```java
// 原生镜像测试配置
@SpringBootTest
@NativeImageTest  // 标记为原生镜像测试
class UserControllerNativeTest {
    
    @Autowired
    private TestRestTemplate restTemplate;
    
    @Test
    void shouldCreateUser() {
        UserCreateRequest request = new UserCreateRequest("native-user", "native@test.com");
        ResponseEntity<UserDTO> response = restTemplate.postForEntity(
            "/api/v1/users", request, UserDTO.class);
        
        assertThat(response.getStatusCode()).isEqualTo(HttpStatus.CREATED);
        assertThat(response.getBody().username()).isEqualTo("native-user");
    }
}
```

### 三、WebTestClient 响应式测试

```java
@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
class UserReactiveControllerTest {
    
    @Autowired
    private WebTestClient webTestClient;
    
    @Test
    void listUsers() {
        webTestClient.get().uri("/api/users")
            .exchange()
            .expectStatus().isOk()
            .expectBodyList(UserDTO.class)
            .hasSize(10)
            .value(users -> {
                assertThat(users.get(0).username()).isNotBlank();
            });
    }
    
    @Test
    void createUser() {
        UserCreateRequest request = new UserCreateRequest("new-user", "new@test.com");
        
        webTestClient.post().uri("/api/users")
            .contentType(MediaType.APPLICATION_JSON)
            .bodyValue(request)
            .exchange()
            .expectStatus().isCreated()
            .expectBody()
            .jsonPath("$.username").isEqualTo("new-user")
            .jsonPath("$.email").isEqualTo("new@test.com");
    }
    
    @Test
    void streamUsers() {
        webTestClient.get().uri("/api/users/stream")
            .accept(MediaType.APPLICATION_NDJSON)
            .exchange()
            .expectStatus().isOk()
            .expectBodyList(UserDTO.class);
    }
}
```

### 四、契约测试（Spring Cloud Contract）

```java
// 契约定义（Groovy DSL）
Contract.make {
    description "should return user by id"
    request {
        method GET()
        urlPath '/api/v1/users/1'
        headers {
            accept(applicationJson())
        }
    }
    response {
        status 200
        body([
            id: 1,
            username: "testuser",
            email: "test@example.com"
        ])
        headers {
            contentType(applicationJson())
        }
    }
}

// 消费者端测试
@SpringBootTest
@AutoConfigureStubRunner(stubsMode = StubRunnerProperties.StubsMode.LOCAL,
    ids = "com.example:user-service:+:stubs:8080")
class UserClientContractTest {
    
    @Autowired
    private UserHttpClient userClient;
    
    @Test
    void shouldGetUserFromProvider() {
        UserDTO user = userClient.getById(1L);
        assertThat(user.username()).isEqualTo("testuser");
    }
}
```

### 五、测试最佳实践（2026）

- **测试容器编排：**
  ```java
  @TestConfiguration
  static class TestContainersConfig {
      @Container
      @ServiceConnection
      static PostgreSQLContainer<?> postgres = new PostgreSQLContainer<>("postgres:16");
      
      @Container
      @ServiceConnection
  ```
