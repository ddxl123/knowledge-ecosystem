# 表单处理

## 核心知识点

### 一、表单类

```python
from django import forms

class ContactForm(forms.Form):
    name = forms.CharField(max_length=100, label='姓名')
    email = forms.EmailField(label='邮箱')
    subject = forms.CharField(max_length=200, label='主题')
    message = forms.CharField(widget=forms.Textarea, label='内容')

    def clean_email(self):
        email = self.cleaned_data['email']
        if not email.endswith('@example.com'):
            raise forms.ValidationError('请使用example.com邮箱')
        return email
```

### 二、模型表单

```python
from django.forms import ModelForm
from .models import Article

class ArticleForm(ModelForm):
    class Meta:
        model = Article
        fields = ['title', 'content', 'category', 'status']
        widgets = {
            'title': forms.TextInput(attrs={'class': 'form-control'}),
            'content': forms.Textarea(attrs={'class': 'form-control', 'rows': 10}),
        }
        labels = {
            'title': '标题',
            'content': '内容',
        }
```

### 三、视图中使用表单

```python
from django.shortcuts import render, redirect
from .forms import ArticleForm

def create_article(request):
    if request.method == 'POST':
        form = ArticleForm(request.POST)
        if form.is_valid():
            article = form.save(commit=False)
            article.author = request.user
            article.save()
            return redirect('articles:detail', slug=article.slug)
    else:
        form = ArticleForm()

    return render(request, 'articles/create.html', {'form': form})
```

### 四、模板中渲染表单

```html
<!-- 方式一：手动渲染 -->
<form method="post">
    {% csrf_token %}

    <div class="form-group">
        <label for="{{ form.title.id_for_label }}">{{ form.title.label }}</label>
        {{ form.title }}
        {% if form.title.errors %}
            <span class="error">{{ form.title.errors.0 }}</span>
        {% endif %}
    </div>

    <button type="submit">提交</button>
</form>

<!-- 方式二：快速渲染 -->
<form method="post">
    {% csrf_token %}
    {{ form.as_p }}
    <button type="submit">提交</button>
</form>
```

### 六、文件上传

```python
class UploadForm(forms.Form):
    file = forms.FileField(label='选择文件')
    description = forms.CharField(max_length=200, required=False)

def upload_file(request):
    if request.method == 'POST':
        form = UploadForm(request.POST, request.FILES)
        if form.is_valid():
            file = form.cleaned_data['file']
            # 保存文件
            with open(f'uploads/{file.name}', 'wb+') as destination:
                for chunk in file.chunks():
                    destination.write(chunk)
            return redirect('upload_success')
    else:
        form = UploadForm()
    return render(request, 'upload.html', {'form': form})
```
