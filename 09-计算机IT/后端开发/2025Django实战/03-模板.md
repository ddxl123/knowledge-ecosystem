### 一、Django 模板语言

```html
<!-- 变量 -->
<h1>{{ article.title }}</h1>
<p>{{ article.content|truncatewords:50 }}</p>
<p>作者：{{ article.author.get_full_name|default:"匿名" }}</p>

<!-- 标签 -->
{% if articles %}
    <ul>
    {% for article in articles %}
        <li class="{% cycle 'odd' 'even' %}">
            <a href="{{ article.get_absolute_url }}">{{ article.title }}</a>
            {% if forloop.first %}<span class="badge">最新</span>{% endif %}
        </li>
    {% empty %}
        <li>暂无文章</li>
    {% endfor %}
    </ul>
{% else %}
    <p>没有找到相关文章。</p>
{% endif %}

<!-- URL 反转 -->
<a href="{% url 'article_detail' pk=article.pk %}">{{ article.title }}</a>

<!-- CSRF 令牌 -->
<form method="post">
    {% csrf_token %}
    {{ form.as_p }}
    <button type="submit">提交</button>
</form>

<!-- 注释 -->
{# 这是模板注释，不会出现在HTML中 #}
```

### 二、模板继承与包含

```html
<!-- base.html - 基础模板 -->
<!DOCTYPE html>
<html>
<head>
    <title>{% block title %}我的网站{% endblock %}</title>
    <link rel="stylesheet" href="{% static 'css/base.css' %}">
    {% block extra_css %}{% endblock %}
</head>
<body>
    {% include 'includes/navbar.html' %}
    
    <main>
        {% block content %}{% endblock %}
    </main>
    
    {% include 'includes/footer.html' %}
    
    <script src="{% static 'js/base.js' %}"></script>
    {% block extra_js %}{% endblock %}
</body>
</html>

<!-- articles/list.html - 继承基础模板 -->
{% extends 'base.html' %}
{% load static %}

{% block title %}文章列表 - {{ block.super }}{% endblock %}

{% block content %}
<div class="container">
    {% for article in articles %}
        {% include 'articles/card.html' with article=article %}
    {% endfor %}
    
    {% include 'includes/pagination.html' with page_obj=page_obj %}
</div>
{% endblock %}

<!-- includes/pagination.html - 被包含的片段 -->
{% if page_obj.has_other_pages %}
<nav>
    {% if page_obj.has_previous %}
        <a href="?page={{ page_obj.previous_page_number }}">上一页</a>
    {% endif %}
    <span>第 {{ page_obj.number }} / {{ page_obj.paginator.num_pages }} 页</span>
    {% if page_obj.has_next %}
        <a href="?page={{ page_obj.next_page_number }}">下一页</a>
    {% endif %}
</nav>
{% endif %}
```

### 三、自定义模板标签与过滤器

```python
# app/templatetags/custom_tags.py
from django import template
from django.utils.safestring import mark_safe
import markdown

register = template.Library()

# 自定义过滤器
@register.filter(name='md')
def markdown_filter(value):
    return mark_safe(markdown.markdown(value, extensions=['codehilite']))

@register.filter
def add_class(field, css_class):
    return field.as_widget(attrs={'class': css_class})

# 简单标签
@register.simple_tag
def current_time(format_string='%Y-%m-%d %H:%M'):
    from datetime import datetime
    return datetime.now().strftime(format_string)

# 包含标签（带模板）
@register.inclusion_tag('includes/tag_list.html')
def show_tags(limit=10):
    from app.models import Tag
    tags = Tag.objects.annotate(count=Count('articles')).order_by('-count')[:limit]
    return {'tags': tags}

# 赋值标签
@register.simple_tag(takes_context=True)
def active_nav(context, url_name):
    request = context['request']
    if request.resolver_match.url_name == url_name:
        return 'active'
    return ''
```

### 四、上下文处理器

```python
# app/context_processors.py
def site_settings(request):
    return {
        'SITE_NAME': '我的网站',
        'SITE_VERSION': '1.0.0',
        'ANALYTICS_ID': 'UA-XXXXX-Y',
    }

def categories(request):
    from .models import Category
    return {
        'nav_categories': Category.objects.filter(is_active=True)[:10]
    }

# 注册（settings.py）
TEMPLATES = [{
    'OPTIONS': {
        'context_processors': [
            'django.template.context_processors.debug',
            'django.template.context_processors.request',
            'django.contrib.auth.context_processors.auth',
            'django.contrib.messages.context_processors.messages',
            'app.context_processors.site_settings',
            'app.context_processors.categories',
        ],
    },
}]
```

### 五、静态文件管理

```python
# settings.py
STATIC_URL = '/static/'
STATIC_ROOT = BASE_DIR / 'staticfiles'         # collectstatic 收集目录
STATICFILES_DIRS = [BASE_DIR / 'static']       # 开发期静态文件目录
STATICFILES_STORAGE = 'django.contrib.staticfiles.storage.ManifestStaticFilesStorage'

MEDIA_URL = '/media/'
MEDIA_ROOT = BASE_DIR / 'media'

# 模板中使用
# {% load static %}
# <img src="{% static 'images/logo.png' %}">
# <img src="{{ article.image.url }}">
```

- **生产环境静态文件：** 使用 Nginx/Apache 直接服务静态文件，或上传至 CDN/OSS
- **WhiteNoise：** 轻量级方案，直接由 Django 服务静态文件
