### 一、Canvas 与渲染模式

```csharp
// Canvas 渲染模式
// Screen Space - Overlay: 覆盖在所有内容上（最常用）
// Screen Space - Camera: 指定摄像机渲染，可被3D物体遮挡
// World Space: 作为3D世界中的物体（血条、名牌）

// Canvas Scaler（适配不同分辨率）
// UI Scale Mode:
// - Constant Pixel Size: 固定像素
// - Scale With Screen Size: 按屏幕大小缩放（推荐）
// - Constant Physical Size: 按物理尺寸缩放

// Reference Resolution: 1920x1080
// Screen Match Mode: Match Width Or Height
// Match: 0.5（宽高各占一半权重）
```

### 二、常用 UI 组件

```csharp
// 文本
using UnityEngine.UI;
using TMPro;

public class UIManager : MonoBehaviour
{
    [SerializeField] private TextMeshProUGUI scoreText;
    [SerializeField] private TextMeshProUGUI healthText;
    [SerializeField] private Slider healthBar;
    [SerializeField] private Image icon;
    [SerializeField] private Button startButton;
    [SerializeField] private InputField nameInput;
    [SerializeField] private Toggle soundToggle;
    [SerializeField] private Dropdown difficultyDropdown;
    
    void Start()
    {
        // 按钮事件
        startButton.onClick.AddListener(OnStartGame);
        
        // 输入框事件
        nameInput.onValueChanged.AddListener(OnNameChanged);
        nameInput.onEndEdit.AddListener(OnNameSubmit);
        
        // 开关事件
        soundToggle.onValueChanged.AddListener(OnSoundToggle);
        
        // 下拉框事件
        difficultyDropdown.onValueChanged.AddListener(OnDifficultyChanged);
    }
    
    public void UpdateScore(int score)
    {
        scoreText.text = $"Score: {score}";
    }
    
    public void UpdateHealth(int current, int max)
    {
        healthText.text = $"{current}/{max}";
        healthBar.value = (float)current / max;
    }
    
    private void OnStartGame() { }
    private void OnNameChanged(string value) { }
    private void OnNameSubmit(string value) { }
    private void OnSoundToggle(bool isOn) { }
    private void OnDifficultyChanged(int index) { }
}
```

### 三、布局系统

```csharp
// Layout Group（自动布局子元素）
// Horizontal Layout Group: 水平排列
// Vertical Layout Group: 垂直排列
// Grid Layout Group: 网格排列

// Content Size Fitter（自适应大小）
// Horizontal Fit / Vertical Fit: Preferred Size / Min Size

// Scroll View（滚动视图）
// Viewport: 可视区域
// Content: 内容容器（挂载 Layout Group）
// Scrollbar: 滚动条

// 动态生成 UI 列表
public class ItemListUI : MonoBehaviour
{
    [SerializeField] private Transform content;
    [SerializeField] private GameObject itemPrefab;
    
    public void RefreshList(List<ItemData> items)
    {
        // 清除旧项
        foreach (Transform child in content)
            Destroy(child.gameObject);
        
        // 生成新项
        foreach (var item in items)
        {
            GameObject go = Instantiate(itemPrefab, content);
            go.GetComponent<ItemUI>().Setup(item);
        }
    }
}
```

### 四、UI 事件系统

```csharp
// EventSystem 组件（场景中必须存在）
// 负责处理 UI 交互事件

// IPointerClickHandler 接口
public class ClickableItem : MonoBehaviour, IPointerClickHandler,
    IPointerEnterHandler, IPointerExitHandler
{
    public void OnPointerClick(PointerEventData eventData)
    {
        if (eventData.button == PointerEventData.InputButton.Left)
            Debug.Log("左键点击");
    }
    
    public void OnPointerEnter(PointerEventData eventData)
    {
        // 鼠标悬停效果
        transform.localScale = Vector3.one * 1.1f;
    }
    
    public void OnPointerExit(PointerEventData eventData)
    {
        transform.localScale = Vector3.one;
    }
}

// 拖拽实现
public class Draggable : MonoBehaviour, IBeginDragHandler, IDragHandler, IEndDragHandler
{
    private Canvas canvas;
    private RectTransform rectTransform;
    
    public void OnBeginDrag(PointerEventData eventData) { }
    
    public void OnDrag(PointerEventData eventData)
    {
        rectTransform.anchoredPosition += eventData.delta / canvas.scaleFactor;
    }
    
    public void OnEndDrag(PointerEventData eventData) { }
}
```

### 五、UI 性能优化

- **减少 Rebuild：** 避免频繁修改 UI 元素属性
- **图集（Atlas）：** 将 UI 图片打成图集，减少 Draw Call
- **对象池：** 列表项使用对象池，避免频繁 Instantiate/Destroy
- **Canvas 分离：** 静态和动态 UI 放在不同 Canvas
- **关闭 Raycast Target：** 不需要交互的 UI 元素关闭 Raycast Target
- **TextMeshPro：** 比 Text 组件性能更好，效果更佳
