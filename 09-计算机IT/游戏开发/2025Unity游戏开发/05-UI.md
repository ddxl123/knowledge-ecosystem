### 二、UGUI组件

```csharp
using UnityEngine.UI;
using TMPro;

public class UIManager : MonoBehaviour
{
    public TextMeshProUGUI scoreText;
    public TextMeshProUGUI healthText;
    public Image healthBar;
    public Slider volumeSlider;
    public Toggle fullscreenToggle;
    public InputField playerNameInput;
    public Button startButton;

    void Start()
    {
        // 文本
        scoreText.text = "Score: 0";

        // 图片
        healthBar.fillAmount = 1f;

        // 滑块
        volumeSlider.onValueChanged.AddListener(OnVolumeChanged);

        // 开关
        fullscreenToggle.onValueChanged.AddListener(OnFullscreenToggle);

        // 输入框
        playerNameInput.onEndEdit.AddListener(OnNameChanged);

        // 按钮
        startButton.onClick.AddListener(OnStartGame);
    }

    void OnVolumeChanged(float value)
    {
        AudioListener.volume = value;
    }

    void OnFullscreenToggle(bool isFullscreen)
    {
        Screen.fullScreen = isFullscreen;
    }

    void OnNameChanged(string name)
    {
        PlayerPrefs.SetString("PlayerName", name);
    }

    void OnStartGame()
    {
        SceneManager.LoadScene("Game");
    }

    // 更新UI
    public void UpdateHealth(int current, int max)
    {
        healthText.text = $"{current}/{max}";
        healthBar.fillAmount = (float)current / max;
    }

    public void UpdateScore(int score)
    {
        scoreText.text = $"Score: {score}";
    }
}
```

### 三、事件系统

```csharp
using UnityEngine.EventSystems;

// 实现拖拽
public class DraggableItem : MonoBehaviour, IBeginDragHandler, IDragHandler, IEndDragHandler
{
    private CanvasGroup canvasGroup;
    private RectTransform rectTransform;

    void Awake()
    {
        canvasGroup = GetComponent<CanvasGroup>();
        rectTransform = GetComponent<RectTransform>();
    }

    public void OnBeginDrag(PointerEventData eventData)
    {
        canvasGroup.alpha = 0.6f;
        canvasGroup.blocksRaycasts = false;
    }

    public void OnDrag(PointerEventData eventData)
    {
        rectTransform.anchoredPosition += eventData.delta;
    }

    public void OnEndDrag(PointerEventData eventData)
    {
        canvasGroup.alpha = 1f;
        canvasGroup.blocksRaycasts = true;
    }
}

// 实现拖放目标
public class DropSlot : MonoBehaviour, IDropHandler
{
    public void OnDrop(PointerEventData eventData)
    {
        if (eventData.pointerDrag != null)
        {
            eventData.pointerDrag.transform.SetParent(transform);
            eventData.pointerDrag.GetComponent<RectTransform>().anchoredPosition = Vector2.zero;
        }
    }
}
```

### 四、自适应布局

```csharp
// 使用锚点（Anchors）实现自适应
// 左上、右上、左下、右下、居中

// Layout组件
// Horizontal Layout Group - 水平布局
// Vertical Layout Group - 垂直布局
// Grid Layout Group - 网格布局

public class DynamicList : MonoBehaviour
{
    public GameObject itemPrefab;
    public Transform content;

    public void PopulateList(List<string> items)
    {
        // 清除现有项目
        foreach (Transform child in content)
        {
            Destroy(child.gameObject);
        }

        // 创建新项目
        foreach (string item in items)
        {
            GameObject newItem = Instantiate(itemPrefab, content);
            newItem.GetComponentInChildren<TextMeshProUGUI>().text = item;
        }
    }
}
```

### 五、弹窗系统

```csharp
public class DialogManager : MonoBehaviour
{
    public GameObject dialogPanel;
    public TextMeshProUGUI titleText;
    public TextMeshProUGUI messageText;
    public Button confirmButton;
    public Button cancelButton;

    private System.Action onConfirm;
    private System.Action onCancel;

    public void ShowDialog(string title, string message,
                          System.Action onConfirm = null,
                          System.Action onCancel = null)
    {
        this.onConfirm = onConfirm;
        this.onCancel = onCancel;

        titleText.text = title;
        messageText.text = message;
        dialogPanel.SetActive(true);
    }

    void Start()
    {
        confirmButton.onClick.AddListener(() =>
        {
            dialogPanel.SetActive(false);
            onConfirm?.Invoke();
        });

        cancelButton.onClick.AddListener(() =>
        {
            dialogPanel.SetActive(false);
            onCancel?.Invoke();
        });
    }
}
```
