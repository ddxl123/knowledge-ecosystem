### 一、Animator 与动画状态机

```csharp
// Animator Controller 结构
// Entry → [Idle] → [Run] → [Jump] → [Attack]
// 参数：Speed(float), IsJumping(bool), Attack(trigger)

public class PlayerAnimator : MonoBehaviour
{
    private Animator animator;
    
    void Awake() => animator = GetComponent<Animator>();
    
    void Update()
    {
        float speed = Mathf.Abs(Input.GetAxis("Horizontal"));
        animator.SetFloat("Speed", speed);
        
        if (Input.GetKeyDown(KeyCode.Space))
            animator.SetBool("IsJumping", true);
        
        if (Input.GetMouseButtonDown(0))
            animator.SetTrigger("Attack");
    }
    
    // 动画事件回调（在动画剪辑中设置）
    public void OnAttackHit()
    {
        // 攻击命中判定
        Collider2D[] hits = Physics2D.OverlapCircleAll(attackPoint.position, attackRadius);
        foreach (var hit in hits)
        {
            hit.GetComponent<IDamageable>()?.TakeDamage(damage);
        }
    }
    
    public void OnFootstep()
    {
        // 脚步声
        audioSource.PlayOneShot(footstepSound);
    }
}
```

### 三、动画过渡控制

```csharp
// Animator Controller 中设置过渡条件
// Has Exit Time: 是否等待当前动画播放完
// Transition Duration: 过渡时间（0.1-0.25秒）
// Conditions: 触发条件（Speed > 0.1, IsJumping == true 等）

// 代码中控制层权重
animator.SetLayerWeight(1, 0.5f); // 设置第二层权重

// 动画状态信息
AnimatorStateInfo stateInfo = animator.GetCurrentAnimatorStateInfo(0);
if (stateInfo.IsName("Attack"))
{
    float normalizedTime = stateInfo.normalizedTime;
    if (normalizedTime >= 1.0f) // 动画播放完成
    {
        // 切换状态
    }
}

// 禁用/启用过渡
animator.applyRootMotion = true; // 应用根运动
```

### 四、2D 精灵动画

```csharp
// Sprite Sheet 动画
// 1. 将序列帧图片拖入创建 Animation
// 2. 在 Sprite Editor 中切片
// 3. 创建 Animation Clip

// 代码控制帧动画
public class SpriteAnimator : MonoBehaviour
{
    [SerializeField] private Sprite[] frames;
    [SerializeField] private float frameRate = 12f;
    
    private SpriteRenderer spriteRenderer;
    private int currentFrame;
    private float timer;
    
    void Update()
    {
        timer += Time.deltaTime;
        if (timer >= 1f / frameRate)
        {
            timer = 0;
            currentFrame = (currentFrame + 1) % frames.Length;
            spriteRenderer.sprite = frames[currentFrame];
        }
    }
}
```

### 五、动画最佳实践

- **状态机设计：** 保持简洁，避免过多状态和过渡
- **混合树：** 用混合树替代多个独立动画，平滑过渡
- **动画事件：** 用动画事件触发音效、特效、伤害判定
- **Root Motion：** 角色动画用根运动，避免滑步
- **动画压缩：** 减少关键帧数量，降低内存占用
- **Avatar Mask：** 上下半身分层动画（跑步+射击）
