# 动画系统

## 核心知识点

### 一、Animator控制器

```csharp
public class PlayerAnimator : MonoBehaviour
{
    private Animator animator;

    void Start()
    {
        animator = GetComponent<Animator>();
    }

    void Update()
    {
        float speed = Input.GetAxis("Vertical");
        float direction = Input.GetAxis("Horizontal");
        bool isJumping = Input.GetButtonDown("Jump");
        bool isAttacking = Input.GetButtonDown("Fire1");

        // 设置动画参数
        animator.SetFloat("Speed", speed);
        animator.SetFloat("Direction", direction);
        animator.SetBool("IsJumping", isJumping);
        animator.SetTrigger("Attack");

        // 播放特定动画
        if (Input.GetKeyDown(KeyCode.R))
        {
            animator.Play("Roll");
        }
    }

    // 动画事件回调
    void OnFootstep()
    {
        // 播放脚步声
        AudioSource.PlayClipAtPoint(footstepSound, transform.position);
    }

    void OnAttackHit()
    {
        // 攻击命中检测
        Collider[] hits = Physics.OverlapSphere(attackPoint.position, attackRange);
        foreach (var hit in hits)
        {
            hit.GetComponent<IDamageable>()?.TakeDamage(damage);
        }
    }
}
```

### 二、动画状态机

```csharp
// Animator Controller设置
// States: Idle, Walk, Run, Jump, Attack, Death
// Transitions: 基于参数的条件转换

// 状态行为脚本
public class AttackState : StateMachineBehaviour
{
    override public void OnStateEnter(Animator animator, AnimatorStateInfo stateInfo, int layerIndex)
    {
        // 进入攻击状态
        animator.GetComponent<Player>().isAttacking = true;
    }

    override public void OnStateExit(Animator animator, AnimatorStateInfo stateInfo, int layerIndex)
    {
        // 离开攻击状态
        animator.GetComponent<Player>().isAttacking = false;
    }
}
```

### 三、骨骼动画

```csharp
// 人形动画重定向
public class AnimationRetarget : MonoBehaviour
{
    public Animator animator;
    public RuntimeAnimatorController controller;

    void Start()
    {
        // 使用人形动画控制器
        animator.runtimeAnimatorController = controller;
    }
}

// IK（反向运动学）
public class IKController : MonoBehaviour
{
    public Animator animator;
    public Transform lookAtTarget;
    public Transform rightHandTarget;

    void OnAnimatorIK(int layerIndex)
    {
        // 头部看向目标
        animator.SetLookAtWeight(1f);
        animator.SetLookAtPosition(lookAtTarget.position);

        // 右手到达目标
        animator.SetIKPositionWeight(AvatarIKGoal.RightHand, 1f);
        animator.SetIKPosition(AvatarIKGoal.RightHand, rightHandTarget.position);
    }
}
```

### 四、粒子系统

```csharp
public class ParticleEffect : MonoBehaviour
{
    public ParticleSystem explosionEffect;
    public ParticleSystem fireEffect;

    void Explode()
    {
        // 播放爆炸特效
        ParticleSystem explosion = Instantiate(explosionEffect, transform.position, Quaternion.identity);
        Destroy(explosion.gameObject, explosion.main.duration);
    }

    void StartFire()
    {
        // 启动火焰
        fireEffect.Play();
    }

    void StopFire()
    {
        // 停止火焰（播放完剩余粒子）
        fireEffect.Stop();
        // 或立即停止
        // fireEffect.Clear();
    }
}
```

### 五、补间动画

```csharp
// 使用DOTween
using DG.Tweening;

public class TweenExample : MonoBehaviour
{
    void Start()
    {
        // 移动
        transform.DOMove(new Vector3(5, 0, 0), 1f)
            .SetEase(Ease.OutBounce)
            .SetLoops(-1, LoopType.Yoyo);

        // 旋转
        transform.DORotate(new Vector3(0, 360, 0), 2f, RotateMode.FastBeyond360);

        // 缩放
        transform.DOScale(Vector3.one * 2f, 0.5f);

        // 颜色
        GetComponent<Renderer>().material.DOColor(Color.red, 1f);

        // 序列动画
        Sequence sequence = DOTween.Sequence();
        sequence.Append(transform.DOMoveX(5, 1f));
        sequence.Append(transform.DOMoveY(3, 1f));
        sequence.Append(transform.DORotate(new Vector3(0, 180, 0), 0.5f));
    }
}
```
