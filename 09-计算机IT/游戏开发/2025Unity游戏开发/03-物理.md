### 一、刚体（Rigidbody）

```csharp
public class PlayerMovement : MonoBehaviour
{
    [SerializeField] private float moveSpeed = 5f;
    [SerializeField] private float jumpForce = 10f;
    
    private Rigidbody2D rb;
    private bool isGrounded;
    
    void Awake() => rb = GetComponent<Rigidbody2D>();
    
    void FixedUpdate()
    {
        // 移动（物理相关操作放 FixedUpdate）
        float horizontal = Input.GetAxis("Horizontal");
        rb.velocity = new Vector2(horizontal * moveSpeed, rb.velocity.y);
        
        // 跳跃
        if (Input.GetKeyDown(KeyCode.Space) && isGrounded)
        {
            rb.AddForce(Vector2.up * jumpForce, ForceMode2D.Impulse);
        }
    }
    
    // 3D 物理
    void FixedUpdate3D()
    {
        Rigidbody rb3D = GetComponent<Rigidbody>();
        rb3D.AddForce(Vector3.forward * 10f, ForceMode.Force);     // 持续力
        rb3D.AddForce(Vector3.up * 5f, ForceMode.Impulse);         // 瞬间力
        rb3D.velocity = new Vector3(horizontal * speed, rb3D.velocity.y, 0);
    }
}
```

### 二、碰撞检测

```csharp
// 碰撞体类型
// BoxCollider / SphereCollider / CapsuleCollider / MeshCollider
// 2D: BoxCollider2D / CircleCollider2D / PolygonCollider2D

public class CollisionHandler : MonoBehaviour
{
    // 物理碰撞（需要 Collider + Rigidbody）
    void OnCollisionEnter2D(Collision2D collision)
    {
        if (collision.gameObject.CompareTag("Enemy"))
        {
            TakeDamage(10);
            // 获取碰撞点
            ContactPoint2D contact = collision.GetContact(0);
            Instantiate(hitEffect, contact.point, Quaternion.identity);
        }
    }
    
    void OnCollisionStay2D(Collision2D collision) { }
    void OnCollisionExit2D(Collision2D collision) { }
    
    // 触发器（Collider 勾选 Is Trigger）
    void OnTriggerEnter2D(Collider2D other)
    {
        if (other.CompareTag("Coin"))
        {
            Destroy(other.gameObject);
            score += 10;
        }
    }
    
    void OnTriggerStay2D(Collider2D other) { }
    void OnTriggerExit2D(Collider2D other) { }
}
```

### 三、射线检测

```csharp
public class Weapon : MonoBehaviour
{
    [SerializeField] private float range = 100f;
    [SerializeField] private LayerMask hitLayers;
    
    void Update()
    {
        if (Input.GetMouseButtonDown(0))
        {
            // 从摄像机发射射线
            Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
            
            if (Physics.Raycast(ray, out RaycastHit hit, range, hitLayers))
            {
                Debug.Log($"Hit: {hit.collider.name} at {hit.point}");
                
                // 对命中目标造成伤害
                IDamageable target = hit.collider.GetComponent<IDamageable>();
                target?.TakeDamage(damage);
                
                // 显示命中特效
                Instantiate(hitEffect, hit.point, Quaternion.LookRotation(hit.normal));
            }
        }
    }
    
    // 球形射线检测（检测范围内敌人）
    void DetectEnemies()
    {
        Collider[] enemies = Physics.OverlapSphere(transform.position, detectionRadius, enemyLayer);
        foreach (Collider enemy in enemies)
        {
            float distance = Vector3.Distance(transform.position, enemy.transform.position);
        }
    }
    
    // 2D 射线
    void Raycast2D()
    {
        RaycastHit2D hit = Physics2D.Raycast(transform.position, Vector2.down, 1f, groundLayer);
        if (hit.collider != null)
        {
            isGrounded = true;
        }
    }
}
```

// 物理材质（控制摩擦力和弹性）
// 在 Inspector 中创建 Physics Material 2D
// Friction: 摩擦力（0=光滑）
// Bounciness: 弹性（0=不弹，1=完全弹性）

// 关节约束
// FixedJoint: 固定连接
// HingeJoint: 铰链（门、钟摆）
// SpringJoint: 弹簧连接
// SliderJoint: 滑动关节

public class ChainLink : MonoBehaviour
{
    void Start()
    {
        HingeJoint2D joint = gameObject.AddComponent<HingeJoint2D>();
        joint.connectedBody = previousLink.GetComponent<Rigidbody2D>();
        joint.limits = new JointAngleLimits2D { min = -30, max = 30 };
        joint.useLimits = true;
    }
}

### 五、物理优化

- **碰撞矩阵：** Edit → Project Settings → Physics 2D → Layer Collision Matrix
  - 禁用不必要的碰撞检测层对
- **FixedUpdate 频率：** 默认 0.02 秒（50Hz），根据需要调整
- **碰撞体简化：** 使用简单碰撞体替代 MeshCollider
- **休眠系统：** 静止物体自动休眠，减少计算
- **插值：** Rigidbody Interpolate 设置为 Interpolate 平滑运动
