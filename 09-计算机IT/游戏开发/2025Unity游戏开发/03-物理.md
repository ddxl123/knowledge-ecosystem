# 物理系统

## 核心知识点

### 一、Rigidbody

```csharp
public class PhysicsMovement : MonoBehaviour
{
    public float moveForce = 10f;
    public float jumpForce = 15f;

    private Rigidbody rb;
    private bool isGrounded;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
        rb.freezeRotation = true; // 防止旋转
    }

    void FixedUpdate()
    {
        // 移动
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");
        Vector3 force = new Vector3(horizontal, 0, vertical) * moveForce;
        rb.AddForce(force);

        // 跳跃
        if (Input.GetButtonDown("Jump") && isGrounded)
        {
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
        }
    }

    // 设置速度
    void SetVelocity(Vector3 velocity)
    {
        rb.linearVelocity = velocity;
    }
}
```

```csharp
        // 持续碰撞
    }

        // 碰撞结束

    // 触发器检测（Collider设为isTrigger）
    void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Pickup"))
        {
            Destroy(other.gameObject);
            AddScore(10);
        }
    }

    void OnTriggerStay(Collider other)
    {
        // 持续触发
    }

    void OnTriggerExit(Collider other)
    {
        // 离开触发器
    }
}
```

### 五、关节

```csharp
public class JointExample : MonoBehaviour
{
    void Start()
    {
        // 铰链关节
        HingeJoint hinge = gameObject.AddComponent<HingeJoint>();
        hinge.axis = Vector3.up;
        hinge.useMotor = true;
        hinge.motor = new JointMotor { targetVelocity = 100, force = 10 };

        // 弹簧关节
        SpringJoint spring = gameObject.AddComponent<SpringJoint>();
        spring.spring = 50;
        spring.damper = 5;
    }
}
```
