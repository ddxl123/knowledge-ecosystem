### 二、移动端适配

```csharp
// 屏幕适配
public class ScreenAdapter : MonoBehaviour
{
    void Start()
    {
        // 设置目标帧率
        Application.targetFrameRate = 60;
        
        // 屏幕方向
        Screen.orientation = ScreenOrientation.LandscapeLeft;
        
        // 安全区域（刘海屏适配）
        Rect safeArea = Screen.safeArea;
        RectTransform rect = GetComponent<RectTransform>();
        Vector2 anchorMin = safeArea.position;
        Vector2 anchorMax = safeArea.position + safeArea.size;
        anchorMin.x /= Screen.width;
        anchorMin.y /= Screen.height;
        anchorMax.x /= Screen.width;
        anchorMax.y /= Screen.height;
        rect.anchorMin = anchorMin;
        rect.anchorMax = anchorMax;
    }
}

// 输入适配
void Update()
{
    // 触摸输入
    if (Input.touchCount > 0)
    {
        Touch touch = Input.GetTouch(0);
        if (touch.phase == TouchPhase.Began)
        {
            Vector3 worldPos = Camera.main.ScreenToWorldPoint(touch.position);
        }
    }
    
    // 虚拟摇杆
    float horizontal = Input.GetAxis("Horizontal"); // 兼容键盘和虚拟摇杆
}
```

```csharp
// 1. 对象池
public class ObjectPool : MonoBehaviour
{
    private Queue<GameObject> pool = new Queue<GameObject>();
    [SerializeField] private GameObject prefab;
    [SerializeField] private int poolSize = 20;
    
    void Start()
    {
        for (int i = 0; i < poolSize; i++)
        {
            GameObject obj = Instantiate(prefab);
            obj.SetActive(false);
            pool.Enqueue(obj);
        }
    }
    
    public GameObject Get()
    {
        GameObject obj = pool.Count > 0 ? pool.Dequeue() : Instantiate(prefab);
        obj.SetActive(true);
        return obj;
    }
    
    public void Return(GameObject obj)
    {
        obj.SetActive(false);
        pool.Enqueue(obj);
    }
}
```

### 四、应用商店发布

- **iOS（App Store）：**
  - 需要 Apple Developer 账号（$99/年）
  - Xcode 构建 IPA 文件
  - App Store Connect 上传
  - 审核周期约 1-3 天

- **Android（Google Play）：**
  - 需要 Google Play Developer 账号（$25 一次性）
  - 构建 AAB 或 APK
  - Google Play Console 上传
  - 审核周期约 1-7 天

- **Steam：**
  - 需要 Steamworks 账号（$100）
  - 使用 Steamworks SDK
  - Steamworks 后台上传
