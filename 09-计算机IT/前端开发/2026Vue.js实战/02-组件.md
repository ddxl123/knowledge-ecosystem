# 组件

## 核心知识点

### 一、组件基础
```vue
<!-- MyComponent.vue -->
<script setup>
import { ref } from 'vue'

// Props
const props = defineProps({
  title: { type: String, required: true },
  count: { type: Number, default: 0 }
})

// Emits
const emit = defineEmits(['update', 'delete'])

function handleClick() {
  emit('update', props.count + 1)
}

// 响应式数据
const message = ref('Hello')
</script>

<template>
  <div>
    <h2>{{ title }}</h2>
    <p>{{ message }}</p>
    <button @click="handleClick">更新 ({{ count }})</button>
  </div>
</template>
```

### 二、组件通信
```vue
<!-- 父组件 -->
<script setup>
import { ref, provide } from 'vue'
import ChildComponent from './ChildComponent.vue'

const message = ref('Hello from parent')

// provide/inject
provide('message', message)
</script>

<template>
  <ChildComponent
    :message="message"
    @update="message = $event"
  />
</template>

<!-- 子组件 -->
<script setup>
import { inject } from 'vue'

const message = inject('message')
</script>
```

### 三、插槽
```vue
<!-- 父组件 -->
<template>
  <CardComponent>
    <template #header>
      <h1>标题</h1>
    </template>

    <template #default="{ item }">
      <p>{{ item.name }}</p>
    </template>

    <template #footer>
      <button>确定</button>
    </template>
  </CardComponent>
</template>

<!-- 子组件 -->
<template>
  <div class="card">
    <div class="header">
      <slot name="header" />
    </div>
    <div class="body">
      <slot :item="currentItem" />
    </div>
    <div class="footer">
      <slot name="footer" />
    </div>
  </div>
</template>
```

### 四、动态组件与异步组件
```vue
<script setup>
import { defineAsyncComponent } from 'vue'

// 动态组件
const currentComponent = computed(() => {
  return activeTab.value === 'A' ? ComponentA : ComponentB
})

// 异步组件
const AsyncComponent = defineAsyncComponent(() =>
  import('./HeavyComponent.vue')
)
</script>

<template>
  <component :is="currentComponent" />
  <Suspense>
    <AsyncComponent />
    <template #fallback>
      <div>加载中...</div>
    </template>
  </Suspense>
</template>
```

### 五、组合式函数（Composables）
```javascript
// useMouse.js
import { ref, onMounted, onUnmounted } from 'vue'

export function useMouse() {
  const x = ref(0)
  const y = ref(0)

  function update(event) {
    x.value = event.pageX
    y.value = event.pageY
  }

  onMounted(() => window.addEventListener('mousemove', update))
  onUnmounted(() => window.removeEventListener('mousemove', update))

  return { x, y }
}

// 使用
const { x, y } = useMouse()
```
