### 二、最短路径算法

```python
import heapq
from collections import defaultdict

# Dijkstra 算法（非负权重，单源最短路径）
def dijkstra(graph, start):
    dist = defaultdict(lambda: float('inf'))
    dist[start] = 0
    pq = [(0, start)]  # (距离, 节点)
    
    while pq:
        d, u = heapq.heappop(pq)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
                heapq.heappush(pq, (dist[v], v))
    return dist

# Bellman-Ford 算法（支持负权重）
def bellman_ford(graph, V, start):
    dist = [float('inf')] * V
    dist[start] = 0
    
    for _ in range(V - 1):
        for u, v, w in graph:  # (起点, 终点, 权重)
            if dist[u] + w < dist[v]:
                dist[v] = dist[u] + w
    
    # 检测负权环
    for u, v, w in graph:
        if dist[u] + w < dist[v]:
            raise ValueError("存在负权环")
    return dist

# Floyd-Warshall 算法（所有节点对最短路径）
def floyd_warshall(graph, V):
    dist = [[float('inf')] * V for _ in range(V)]
    for i in range(V):
        dist[i][i] = 0
    for u, v, w in graph:
        dist[u][v] = w
    
    for k in range(V):
        for i in range(V):
            for j in range(V):
                dist[i][j] = min(dist[i][j], dist[i][k] + dist[k][j])
    return dist
```

### 三、最小生成树

```python
# Prim 算法（贪心，从一个节点出发）
def prim(graph, V):
    visited = set()
    min_edge = [(0, 0)]  # (权重, 节点)
    total_cost = 0
    
    while len(visited) < V:
        w, u = heapq.heappop(min_edge)
        if u in visited:
            continue
        visited.add(u)
        total_cost += w
        for v, weight in graph[u]:
            if v not in visited:
                heapq.heappush(min_edge, (weight, v))
    return total_cost

# Kruskal 算法（并查集）
class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n
    
    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]
    
    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py: return False
        if self.rank[px] < self.rank[py]: px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]: self.rank[px] += 1
        return True

def kruskal(edges, V):
    edges.sort(key=lambda x: x[2])  # 按权重排序
    uf = UnionFind(V)
    mst_cost = 0
    for u, v, w in edges:
        if uf.union(u, v):
            mst_cost += w
    return mst_cost
```

### 四、拓扑排序

```python
from collections import deque

# Kahn 算法（BFS 方式）
def topological_sort(graph, V):
    in_degree = [0] * V
    for u in graph:
        for v in graph[u]:
            in_degree[v] += 1
    
    queue = deque([i for i in range(V) if in_degree[i] == 0])
    result = []
    
    while queue:
        u = queue.popleft()
        result.append(u)
        for v in graph[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    
    if len(result) != V:
        raise ValueError("图中存在环，无法拓扑排序")
    return result

# DFS 方式
def topo_dfs(graph, V):
    visited = [False] * V
    stack = []
    
    def dfs(u):
        visited[u] = True
        for v in graph[u]:
            if not visited[v]:
                dfs(v)
        stack.append(u)
    
    for i in range(V):
        if not visited[i]:
            dfs(i)
    return stack[::-1]
```

### 五、应用场景

- **Dijkstra：** 导航最短路径、网络路由
- **Bellman-Ford：** 含负权边的最短路径（如套利检测）
- **Floyd：** 任意两点间最短路径（如距离矩阵）
- **Prim/Kruskal：** 网络布线、聚类分析
- **拓扑排序：** 任务调度、课程安排、编译依赖
- **BFS：** 社交网络最短关系链
- **DFS：** 连通分量检测、迷宫求解
