### 二、概率分布

```python
from scipy import stats

# 正态分布
norm = stats.norm(loc=100, scale=15)  # 均值100，标准差15
norm.pdf(100)       # 概率密度
norm.cdf(130)       # 累积概率 P(X ≤ 130)
norm.ppf(0.95)      # 分位数
norm.rvs(1000)      # 随机抽样

# 二项分布
binom = stats.binom(n=10, p=0.5)
binom.pmf(5)        # P(X=5)
binom.cdf(5)        # P(X ≤ 5)

# 泊松分布（适合计数数据）
poisson = stats.poisson(mu=3)
poisson.pmf(5)      # P(X=5)

# 检验正态性
stat, p_value = stats.shapiro(data)
print(f'Shapiro-Wilk p值: {p_value:.4f}')  # p > 0.05 则不能拒绝正态假设
```

### 三、假设检验

```python
# 单样本 t 检验（均值是否等于某值）
t_stat, p_value = stats.ttest_1samp(data, popmean=50000)
print(f't={t_stat:.3f}, p={p_value:.4f}')
# p < 0.05 → 拒绝 H0，均值显著不等于50000

# 双样本 t 检验（两组均值是否相等）
group_a = df[df['department'] == '技术']['salary']
group_b = df[df['department'] == '销售']['salary']
t_stat, p_value = stats.ttest_ind(group_a, group_b, equal_var=False)

# 配对 t 检验（前后对比）
t_stat, p_value = stats.ttest_rel(before, after)

# 卡方检验（分类变量独立性）
contingency = pd.crosstab(df['gender'], df['department'])
chi2, p_value, dof, expected = stats.chi2_contingency(contingency)

# ANOVA（多组均值比较）
groups = [df[df['city'] == city]['salary'] for city in df['city'].unique()]
f_stat, p_value = stats.f_oneway(*groups)
```

### 四、相关分析

```python
# 皮尔逊相关系数（线性关系）
corr_matrix = df[['age', 'salary', 'experience']].corr(method='pearson')
print(corr_matrix)

# 斯皮尔曼相关系数（单调关系，适合非正态数据）
corr, p_value = stats.spearmanr(df['age'], df['salary'])

# 可视化相关矩阵
import seaborn as sns
sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', center=0, vmin=-1, vmax=1)

# 相关系数解读
# |r| < 0.3: 弱相关
# 0.3 ≤ |r| < 0.7: 中等相关
# |r| ≥ 0.7: 强相关
# 注意：相关不等于因果！
```

### 五、回归分析

```python
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score, mean_squared_error
import statsmodels.api as sm

# 简单线性回归
X = df[['experience']].values
y = df['salary'].values

model = LinearRegression()
model.fit(X, y)
print(f'斜率: {model.coef_[0]:.2f}, 截距: {model.intercept_:.2f}')
print(f'R²: {model.score(X, y):.4f}')

# 多元回归（statsmodels，含统计检验）
X = sm.add_constant(df[['experience', 'age', 'education']])
model = sm.OLS(y, X).fit()
print(model.summary())  # 详细统计报告

# 回归诊断
# R²: 模型解释力（越接近1越好）
# p值: 系数是否显著（< 0.05 显著）
# 残差分析: 检查线性假设、异方差、正态性
```
