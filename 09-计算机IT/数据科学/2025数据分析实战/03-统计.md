# 统计分析

## 核心知识点

### 一、描述统计

```python
import numpy as np
from scipy import stats

data = [23, 25, 28, 30, 32, 35, 38, 40, 42, 45]

# 集中趋势
mean = np.mean(data)        # 均值
median = np.median(data)    # 中位数
mode = stats.mode(data)     # 众数

# 离散程度
std = np.std(data)          # 标准差
var = np.var(data)          # 方差
data_range = np.ptp(data)   # 极差
q1, q3 = np.percentile(data, [25, 75])  # 四分位数
iqr = q3 - q1              # 四分位距

# 分布形态
skewness = stats.skew(data)     # 偏度
kurtosis = stats.kurtosis(data) # 峰度
```

### 二、假设检验

```python
# t检验（两组均值比较）
group1 = [85, 88, 90, 92, 87]
group2 = [78, 82, 85, 80, 76]

t_stat, p_value = stats.ttest_ind(group1, group2)
print(f"t-statistic: {t_stat}, p-value: {p_value}")
# p < 0.05 → 拒绝原假设，两组有显著差异

# 配对t检验
before = [85, 88, 90, 92, 87]
after = [90, 92, 95, 96, 93]
t_stat, p_value = stats.ttest_rel(after, before)

# 卡方检验
observed = [[50, 30, 20], [35, 40, 25]]
chi2, p_value, dof, expected = stats.chi2_contingency(observed)

# ANOVA（多组比较）
group_a = [85, 88, 90, 92, 87]
group_b = [78, 82, 85, 80, 76]
group_c = [90, 92, 95, 96, 93]
f_stat, p_value = stats.f_oneway(group_a, group_b, group_c)
```

### 四、回归分析

```python
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score, mean_squared_error

# 线性回归
X = df[['age', 'experience']].values
y = df['salary'].values

model = LinearRegression()
model.fit(X, y)

print(f"Coefficients: {model.coef_}")
print(f"Intercept: {model.intercept_}")

y_pred = model.predict(X)
print(f"R² Score: {r2_score(y, y_pred)}")
print(f"RMSE: {np.sqrt(mean_squared_error(y, y_pred))}")

# 统计学回归
import statsmodels.api as sm

X_with_const = sm.add_constant(X)
ols_model = sm.OLS(y, X_with_const).fit()
print(ols_model.summary())
```

### 五、概率分布

```python
# 正态分布
normal_dist = stats.norm(loc=100, scale=15)
pdf_value = normal_dist.pdf(115)    # 概率密度
cdf_value = normal_dist.cdf(115)    # 累积概率
sample = normal_dist.rvs(size=1000) # 随机采样

# 二项分布
binom_dist = stats.binom(n=10, p=0.5)
prob = binom_dist.pmf(5)  # P(X=5)

# 泊松分布
poisson_dist = stats.poisson(mu=3)
prob = poisson_dist.pmf(2)  # P(X=2)

# 正态性检验
stat, p_value = stats.shapiro(data)
# p > 0.05 → 数据符合正态分布
```
