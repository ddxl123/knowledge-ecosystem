```python
# 用户留存分析
def retention_analysis(df, user_col, date_col, periods=7):
    df[date_col] = pd.to_datetime(df[date_col])
    first_visit = df.groupby(user_col)[date_col].min().reset_index()
    first_visit.columns = [user_col, 'first_date']
    df = df.merge(first_visit, on=user_col)
    df['days_since'] = (df[date_col] - df['first_date']).dt.days
    retention = df.groupby('days_since')[user_col].nunique()
    return retention / retention.iloc[0]

# RFM 用户分层
def rfm_score(df):
    snapshot = df['order_date'].max() + pd.Timedelta(days=1)
    rfm = df.groupby('user_id').agg({
        'order_date': lambda x: (snapshot - x.max()).days,  # Recency
        'order_id': 'count',                                 # Frequency
        'amount': 'sum'                                      # Monetary
    })
    rfm.columns = ['R', 'F', 'M']
    # 打分（1-5分）
    for col in ['R', 'F', 'M']:
        rfm[f'{col}_score'] = pd.qcut(rfm[col], 5, labels=[5,4,3,2,1] if col=='R' else [1,2,3,4,5])
    return rfm
```

```python
cumsum = product_sales.cumsum() / product_sales.sum()
product_sales = pd.DataFrame({'sales': product_sales, 'cum_pct': cumsum})
product_sales['abc'] = pd.cut(product_sales['cum_pct'], bins=[0, 0.8, 0.95, 1.0], labels=['A', 'B', 'C'])
```

### 四、A/B 测试

```python
from scipy import stats

# A/B 测试分析
def ab_test(control, treatment, alpha=0.05):
    # 转化率比较
    control_rate = control.sum() / len(control)
    treatment_rate = treatment.sum() / len(treatment)
    
    # 卡方检验
    contingency = [[control.sum(), len(control) - control.sum()],
                   [treatment.sum(), len(treatment) - treatment.sum()]]
    chi2, p_value, dof, expected = stats.chi2_contingency(contingency)
    
    lift = (treatment_rate - control_rate) / control_rate
    
    result = {
        'control_rate': f'{control_rate:.2%}',
        'treatment_rate': f'{treatment_rate:.2%}',
        'lift': f'{lift:.2%}',
        'p_value': f'{p_value:.4f}',
        'significant': p_value < alpha
    }
    return result

# 样本量计算
from statsmodels.stats.power import NormalIndPower
power_analysis = NormalIndPower()
sample_size = power_analysis.solve_power(
    effect_size=0.05,  # 最小可检测效应
    alpha=0.05,
    power=0.8,
    alternative='two-sided'
)
```
