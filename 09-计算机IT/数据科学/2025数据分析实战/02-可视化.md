### 一、Matplotlib

```python
import matplotlib.pyplot as plt

# 折线图
plt.figure(figsize=(10, 6))
plt.plot(dates, sales, marker='o', color='blue', label='Sales')
plt.plot(dates, costs, marker='s', color='red', label='Costs')
plt.title('Sales vs Costs')
plt.xlabel('Date')
plt.ylabel('Amount')
plt.legend()
plt.grid(True)
plt.savefig('chart.png', dpi=150)
plt.show()

# 柱状图
plt.bar(categories, values, color=['#2196F3', '#4CAF50', '#FF9800'])

# 饼图
plt.pie(sizes, labels=labels, autopct='%1.1f%%', startangle=90)

# 子图
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
axes[0, 0].plot(x, y1)
axes[0, 1].bar(x, y2)
axes[1, 0].scatter(x, y3)
axes[1, 1].hist(y4, bins=20)
```


### 三、Plotly（交互式）

```python
import plotly.express as px
import plotly.graph_objects as go

# 交互式折线图
fig = px.line(df, x='date', y='sales', color='product',
              title='Sales Trend')
fig.show()

# 交互式散点图
fig = px.scatter(df, x='gdp', y='life_expectancy',
                 size='population', color='continent',
                 hover_name='country',
                 title='World Development')
fig.show()

# 3D图表
fig = go.Figure(data=[go.Scatter3d(
    x=df['x'], y=df['y'], z=df['z'],
    mode='markers',
    marker=dict(size=5, color=df['z'], colorscale='Viridis')
)])

# 地图
fig = px.choropleth(df, locations='country_code',
                    color='gdp', hover_name='country')
fig.show()
```

### 四、Altair

```python
import altair as alt

# 基本图表
chart = alt.Chart(df).mark_bar().encode(
    x='category:N',
    y='sum(sales):Q',
    color='category:N'
)

# 交互式选择
brush = alt.selection_interval()
points = alt.Chart(df).mark_point().encode(
    x='age:Q',
    y='salary:Q',
    color=alt.condition(brush, 'department:N', alt.value('lightgray'))
).add_selection(brush)

bars = alt.Chart(df).mark_bar().encode(
    y='department:N',
    color='department:N',
    x='count():Q'
).transform_filter(brush)
```
