## 核心知识点

### 一、PDO 连接

```php
<?php
// PDO 连接
try {
    $dsn = "mysql:host=localhost;dbname=mydb;charset=utf8mb4";
    $options = [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES => false,
    ];
    $pdo = new PDO($dsn, 'root', 'password', $options);
} catch (PDOException $e) {
    die("数据库连接失败: " . $e->getMessage());
}

// 查询测试
$stmt = $pdo->query("SELECT VERSION()");
echo $stmt->fetchColumn();
```

### 二、CRUD 操作

```php
// 查询（SELECT）
$stmt = $pdo->query("SELECT * FROM users WHERE status = 'active'");
$users = $stmt->fetchAll();
// 或逐行获取
while ($row = $stmt->fetch()) {
    echo $row['username'];
}

// 查询单条
$stmt = $pdo->prepare("SELECT * FROM users WHERE id = :id");
$stmt->execute(['id' => 1]);
$user = $stmt->fetch();

// 插入（INSERT）
$stmt = $pdo->prepare("INSERT INTO users (username, email) VALUES (:username, :email)");
$stmt->execute([
    'username' => 'alice',
    'email' => 'alice@example.com'
]);
echo "新记录ID: " . $pdo->lastInsertId();

// 更新（UPDATE）
$stmt = $pdo->prepare("UPDATE users SET email = :email WHERE id = :id");
$stmt->execute(['email' => 'new@example.com', 'id' => 1]);
echo "影响行数: " . $stmt->rowCount();

// 删除（DELETE）
$stmt = $pdo->prepare("DELETE FROM users WHERE id = :id");
$stmt->execute(['id' => 1]);
```

### 三、预处理语句（防 SQL 注入）

```php
// 命名参数
$stmt = $pdo->prepare("SELECT * FROM users WHERE username = :username AND status = :status");
$stmt->execute(['username' => $input_username, 'status' => 'active']);

// 位置参数
$stmt = $pdo->prepare("SELECT * FROM users WHERE username = ? AND status = ?");
$stmt->execute([$input_username, 'active']);

// LIKE 查询
$stmt = $pdo->prepare("SELECT * FROM users WHERE username LIKE :keyword");
$stmt->execute(['keyword' => "%{$keyword}%"]);

// IN 查询
$ids = [1, 2, 3, 4, 5];
$placeholders = implode(',', array_fill(0, count($ids), '?'));
$stmt = $pdo->prepare("SELECT * FROM users WHERE id IN ($placeholders)");
$stmt->execute($ids);
```

### 四、事务处理

```php
try {
    $pdo->beginTransaction();
    
    // 转账操作
    $stmt = $pdo->prepare("UPDATE accounts SET balance = balance - :amount WHERE id = :from");
    $stmt->execute(['amount' => 100, 'from' => 1]);
    
    $stmt = $pdo->prepare("UPDATE accounts SET balance = balance + :amount WHERE id = :to");
    $stmt->execute(['amount' => 100, 'to' => 2]);
    
    $pdo->commit();
    echo "转账成功";
} catch (Exception $e) {
    $pdo->rollBack();
    echo "转账失败: " . $e->getMessage();
}
```

### 五、Eloquent ORM（Laravel）

```php
// 模型定义
class User extends Model
{
    protected $table = 'users';
    protected $fillable = ['username', 'email', 'password'];
    protected $hidden = ['password'];
    
    public function posts()
    {
        return $this->hasMany(Post::class);
    }
}

// 查询
$users = User::where('status', 'active')
    ->orderBy('created_at', 'desc')
    ->paginate(20);

$user = User::find(1);
$user = User::where('email', 'test@example.com')->first();

// 创建
$user = User::create([
    'username' => 'alice',
    'email' => 'alice@example.com',
    'password' => Hash::make('password')
]);

// 更新
$user->update(['email' => 'new@example.com']);

// 删除
$user->delete();

// 关联查询
$user->posts()->where('status', 'published')->get();
```
