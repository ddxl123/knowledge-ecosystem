# 协程编程

## 核心知识点

### 一、协程基础

#### 1. 启动协程
```kotlin
import kotlinx.coroutines.*

fun main() = runBlocking {
    launch {
        delay(1000L)
        println("World!")
    }
    println("Hello,")
}

// GlobalScope启动
GlobalScope.launch {
    delay(1000L)
    println("Background task")
}
```

#### 2. 挂起函数
```kotlin
suspend fun fetchUser(): User {
    delay(1000L) // 模拟网络请求
    return User("Alice", "alice@example.com")
}

suspend fun fetchPosts(userId: String): List<Post> {
    delay(1500L)
    return listOf(Post("Title 1"), Post("Title 2"))
}
```

### 二、协程构建器

#### 1. launch（不返回结果）
```kotlin
val job = CoroutineScope(Dispatchers.IO).launch {
    val user = fetchUser()
    println("User: ${user.name}")
}

job.join() // 等待完成
job.cancel() // 取消
```

#### 2. async（返回结果）
```kotlin
CoroutineScope(Dispatchers.IO).async {
    val user = fetchUser()
    user
}.await() // 获取结果

// 并发执行
val userDeferred = async { fetchUser() }
val postsDeferred = async { fetchPosts("1") }

val user = userDeferred.await()
val posts = postsDeferred.await()
```

#### 3. withContext（切换上下文）
```kotlin
suspend fun loadData(): Data {
    return withContext(Dispatchers.IO) {
        // 在IO线程执行
        fetchDataFromNetwork()
    }
}
```

### 三、Flow

#### 1. 基本Flow
```kotlin
fun numbers(): Flow<Int> = flow {
    for (i in 1..3) {
        delay(100)
        emit(i)
    }
}

// 收集
numbers().collect { value ->
    println(value)
}
```

#### 2. Flow操作符
```kotlin
numbers()
    .map { it * it }
    .filter { it > 1 }
    .take(2)
    .collect { println(it) }

// 组合多个Flow
val flow1 = flowOf(1, 2, 3)
val flow2 = flowOf("a", "b", "c")

flow1.zip(flow2) { a, b -> "$a$b" }
    .collect { println(it) } // 1a, 2b, 3c
```

#### 3. StateFlow和SharedFlow
```kotlin
class UserViewModel : ViewModel() {
    private val _users = MutableStateFlow<List<User>>(emptyList())
    val users: StateFlow<List<User>> = _users.asStateFlow()

    private val _events = MutableSharedFlow<Event>()
    val events: SharedFlow<Event> = _events.asSharedFlow()

    fun loadUsers() {
        viewModelScope.launch {
            _users.value = repository.getUsers()
        }
    }
}
```

### 四、异常处理

```kotlin
// try-catch
try {
    val result = riskyOperation()
} catch (e: Exception) {
    println("Error: ${e.message}")
}

// CoroutineExceptionHandler
val handler = CoroutineExceptionHandler { _, exception ->
    println("Caught: ${exception.message}")
}

CoroutineScope(Dispatchers.IO + handler).launch {
    throw RuntimeException("Something went wrong")
}

// Flow异常处理
numbers()
    .catch { e -> println("Flow error: ${e.message}") }
    .collect { println(it) }

// supervisorScope
supervisorScope {
    launch { throw Exception("Child 1 failed") }
    launch { println("Child 2 still runs") }
}
```

### 五、结构化并发

```kotlin
// 父协程等待所有子协程
fun main() = runBlocking {
    val parentJob = launch {
        val child1 = launch {
            delay(1000)
            println("Child 1 done")
        }
        val child2 = launch {
            delay(2000)
            println("Child 2 done")
        }
    }
    parentJob.join() // 等待所有子协程完成
    println("All done")
}

// 取消传播
val parentJob = launch {
    launch {
        delay(1000)
        println("This won't print")
    }
}
parentJob.cancel() // 子协程也被取消
```
