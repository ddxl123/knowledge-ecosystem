# Kotlin基础语法

## 核心知识点

### 一、变量与类型

```kotlin
// 可变变量
var name = "Alice"
name = "Bob"

// 不可变变量
val age = 25

// 类型声明
val score: Double = 95.5
val isActive: Boolean = true
val greeting: String = "Hello"

// 字符串模板
val message = "Hello, $name! You are $age years old."
val calculation = "Result: ${age * 2}"
```

### 二、空安全

```kotlin
// 非空类型
var name: String = "Alice"
// name = null // 编译错误

// 可空类型
var nullableName: String? = "Alice"
nullableName = null

// 安全调用
val length = nullableName?.length

// Elvis运算符
val displayName = nullableName ?: "Unknown"

// 非空断言
val definiteLength = nullableName!!.length

// 安全转换
val obj: Any = "Hello"
val str: String? = obj as? String

// let安全调用
nullableName?.let { name ->
    println("Name: $name, Length: ${name.length}")
}
```

### 三、函数

```kotlin
// 基本函数
fun add(a: Int, b: Int): Int {
    return a + b
}

// 单表达式函数
fun multiply(a: Int, b: Int) = a * b

// 默认参数
fun greet(name: String, greeting: String = "Hello"): String {
    return "$greeting, $name!"
}

// 命名参数
greet(name = "Alice", greeting = "Hi")

// 可变参数
fun sum(vararg numbers: Int): Int {
    return numbers.sum()
}

// 扩展函数
fun String.isPalindrome(): Boolean {
    return this == this.reversed()
}

"racecar".isPalindrome() // true
```

### 四、类与对象

```kotlin
// 主构造函数
class Person(val name: String, var age: Int) {
    init {
        println("Created person: $name")
    }

    fun greet() = "Hi, I'm $name"
}

// 数据类
data class User(val name: String, val email: String)

// 密封类
sealed class Result<out T> {
    data class Success<T>(val data: T) : Result<T>()
    data class Error(val message: String) : Result<Nothing>()
    object Loading : Result<Nothing>()
}

// 对象声明（单例）
object Database {
    fun connect() { /* ... */ }
}

// 伴生对象
class MyClass {
    companion object {
        fun create(): MyClass = MyClass()
    }
}
```

### 五、集合操作

```kotlin
val numbers = listOf(1, 2, 3, 4, 5)
val mutableList = mutableListOf("a", "b", "c")
val map = mapOf("name" to "Alice", "age" to 25)

// 高阶函数
val doubled = numbers.map { it * 2 }
val evens = numbers.filter { it % 2 == 0 }
val sum = numbers.reduce { acc, i -> acc + i }
val grouped = numbers.groupBy { if (it % 2 == 0) "even" else "odd" }

// 链式调用
val result = numbers
    .filter { it > 2 }
    .map { it * 10 }
    .take(2)

// 解构
val (first, second, third) = listOf(1, 2, 3)
```

### 六、when表达式

```kotlin
val result = when (x) {
    1 -> "One"
    2, 3 -> "Two or Three"
    in 4..10 -> "Four to Ten"
    is Int -> "Some other int"
    else -> "Unknown"
}

// when作为函数体
fun describe(obj: Any): String = when (obj) {
    is Int -> "Integer: $obj"
    is String -> "String of length ${obj.length}"
    is List<*> -> "List of size ${obj.size}"
    else -> "Unknown"
}
```
