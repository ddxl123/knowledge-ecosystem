# Swift基础语法

## 核心知识点

### 一、变量与常量

```swift
// 变量
var name = "Alice"
name = "Bob"

// 常量
let pi = 3.14159

// 类型标注
var age: Int = 25
var score: Double = 95.5
var isActive: Bool = true
var greeting: String = "Hello"
```

### 二、可选类型

```swift
// 叚可选值
var optionalName: String? = "Alice"
optionalName = nil

// 强制解包（危险）
let name = optionalName!

// 安全解包
if let name = optionalName {
    print("Name: \(name)")
}

// guard语句
func greet(name: String?) {
    guard let name = name else {
        print("No name provided")
        return
    }
    print("Hello, \(name)")
}

// 空合运算符
let displayName = optionalName ?? "Unknown"

// 可选链
let count = optionalName?.count
```

### 三、控制流

```swift
// if-else
if score >= 90 {
    grade = "A"
} else if score >= 80 {
    grade = "B"
} else {
    grade = "C"
}

// switch
switch number {
case 0:
    print("Zero")
case 1...9:
    print("Single digit")
case 10...99:
    print("Double digits")
default:
    print("Larger")
}

// for-in
for i in 0..<10 {
    print(i)
}

for (index, value) in array.enumerated() {
    print("\(index): \(value)")
}

// while / repeat-while
while condition {
    // ...
}
```

### 四、函数

```swift
// 基本函数
func add(_ a: Int, _ b: Int) -> Int {
    return a + b
}

// 参数标签
func greet(person name: String, from hometown: String) -> String {
    return "Hello \(name) from \(hometown)"
}
greet(person: "Alice", hometown: "Beijing")

// 默认参数
func power(base: Int, exponent: Int = 2) -> Int {
    return Int(pow(Double(base), Double(exponent)))
}

// 可变参数
func average(_ numbers: Double...) -> Double {
    let total = numbers.reduce(0, +)
    return total / Double(numbers.count)
}

// 返回多个值
func minMax(array: [Int]) -> (min: Int, max: Int)? {
    guard !array.isEmpty else { return nil }
    return (array.min()!, array.max()!)
}
```

### 五、闭包

```swift
// 闭包表达式
let multiply = { (a: Int, b: Int) -> Int in
    return a * b
}

// 尾随闭包
let sorted = names.sorted { $0 < $1 }

// 闭包作为参数
func fetchData(completion: @escaping (Result<Data, Error>) -> Void) {
    // 异步操作
}

// 捕获值
func makeCounter() -> () -> Int {
    var count = 0
    return {
        count += 1
        return count
    }
}
```

### 六、集合类型

```swift
// 数组
var fruits = ["apple", "banana", "cherry"]
fruits.append("date")
fruits.remove(at: 0)

// 字典
var scores = ["Alice": 95, "Bob": 87]
scores["Charlie"] = 92

// 集合
var uniqueNumbers: Set = [1, 2, 3, 3, 4] // {1, 2, 3, 4}

// 枚举
enum Direction {
    case north, south, east, west
}

enum Result<T> {
    case success(T)
    case failure(Error)
}
```
