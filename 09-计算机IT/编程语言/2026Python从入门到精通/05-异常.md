## 核心知识点

### 一、异常捕获
```python
# try-except
try:
    result = 10 / 0
except ZeroDivisionError:
    print("不能除以零")

# 捕获多个异常
try:
    result = int("abc")
except (ValueError, TypeError) as e:
    print(f"错误: {e}")

# 捕获所有异常
try:
    pass
except Exception as e:
    print(f"错误: {e}")

# try-except-else-finally
try:
    result = 10 / 2
except ZeroDivisionError:
    print("除以零")
else:
    print(f"结果: {result}")  # 没有异常时执行
finally:
    print("总是执行")  # 总是执行
```

### 二、常见内置异常
```python
# ValueError: 值错误
int("abc")

# TypeError: 类型错误
"hello" + 1

# IndexError: 索引错误
lst = [1, 2, 3]
lst[10]

# KeyError: 键错误
d = {"a": 1}
d["b"]

# AttributeError: 属性错误
"hello".append("!")

# FileNotFoundError: 文件未找到
open("nonexistent.txt")

# ImportError: 导入错误
import nonexistent_module

# NameError: 名称错误
print(undefined_variable)
```

### 三、自定义异常
```python
# 自定义异常类
class CustomError(Exception):
    def __init__(self, message, code):
        super().__init__(message)
        self.code = code

# 使用自定义异常
def validate_age(age):
    if age < 0:
        raise CustomError("年龄不能为负数", 1001)
    if age > 150:
        raise CustomError("年龄不合理", 1002)

try:
    validate_age(-1)
except CustomError as e:
    print(f"错误代码: {e.code}, 错误信息: {e}")
```

### 四、异常最佳实践
```python
# 1. 具体异常优于宽泛异常
try:
    pass
except ValueError:  # 好
    pass
except Exception:   # 不推荐
    pass

# 2. 不要忽略异常
try:
    pass
except Exception:
    pass  # 不推荐，至少记录日志

# 3. 使用上下文管理器
with open("file.txt") as f:
    content = f.read()

# 4. 异常链
try:
    pass
except ValueError as e:
    raise RuntimeError("处理失败") from e

# 5. 使用assert进行调试
assert x > 0, "x必须为正数"
```
