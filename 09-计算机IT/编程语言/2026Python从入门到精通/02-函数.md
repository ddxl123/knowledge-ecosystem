## 核心知识点

### 一、函数定义与调用
```python
# 定义函数
def greet(name):
    """向某人打招呼"""
    return f"Hello, {name}!"

# 调用函数
message = greet("Alice")
```

### 二、参数类型
```python
# 位置参数
def add(a, b):
    return a + b

# 默认参数
def greet(name, greeting="Hello"):
    return f"{greeting}, {name}!"

# 可变参数 *args
def sum_all(*args):
    return sum(args)

# 关键字参数 **kwargs
def print_info(**kwargs):
    for key, value in kwargs.items():
        print(f"{key}: {value}")

# 参数组合顺序：位置参数，默认参数，*args，**kwargs
```

### 三、返回值
```python
# 返回单个值
def square(x):
    return x ** 2

# 返回多个值
def min_max(numbers):
    return min(numbers), max(numbers)

# 返回None
def print_hello():
    print("Hello")
    # 隐式返回None
```

### 四、作用域
```python
# 局部变量
def func():
    x = 10  # 局部变量

# 全局变量
y = 20  # 全局变量

def func():
    global y  # 声明使用全局变量
    y = 30

# 闭包
def outer():
    x = 10
    def inner():
        return x
    return inner
```

### 五、Lambda函数
```python
# Lambda表达式
square = lambda x: x ** 2

# 常用高阶函数
numbers = [1, 2, 3, 4, 5]
squared = list(map(lambda x: x**2, numbers))
evens = list(filter(lambda x: x % 2 == 0, numbers))
total = reduce(lambda x, y: x + y, numbers)
```

### 六、装饰器
```python
# 装饰器
def timer(func):
    import time
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        print(f"函数执行时间: {end - start}秒")
        return result
    return wrapper

@timer
def slow_function():
    import time
    time.sleep(1)
```
