### 一、模块基础
```python
# 导入模块
import math
import os.path
from datetime import datetime
from collections import defaultdict, namedtuple

# 使用模块
math.sqrt(16)
datetime.now()
```

### 二、自定义模块
```python
# mymodule.py
def greet(name):
    return f"Hello, {name}!"

class Calculator:
    def add(self, a, b):
        return a + b

PI = 3.14159

# 使用自定义模块
import mymodule
mymodule.greet("Alice")
```

### 三、包的创建
```
myproject/
    __init__.py
    module1.py
    module2.py
    subpackage/
        __init__.py
        module3.py
```

```python
# __init__.py
from .module1 import func1
from .module2 import func2

# 使用包
from myproject import func1
from myproject.subpackage import module3
```

### 四、常用标准库
```python
# os: 操作系统接口
import os
os.getcwd()
os.listdir(".")

# sys: 系统相关
import sys
sys.argv
sys.path

# datetime: 日期时间
from datetime import datetime, timedelta
now = datetime.now()
tomorrow = now + timedelta(days=1)

# re: 正则表达式
import re
pattern = r"\d+"
result = re.findall(pattern, "abc123def456")

# collections: 高级数据结构
from collections import Counter, defaultdict, namedtuple
Counter([1, 2, 2, 3, 3, 3])

# itertools: 迭代器工具
from itertools import combinations, permutations, product
combinations([1, 2, 3], 2)

# functools: 函数工具
from functools import lru_cache, partial
@lru_cache(maxsize=128)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n-1) + fibonacci(n-2)
```

### 六、常用第三方库
```python

# numpy: 数值计算
import numpy as np

np.mean(arr)

# pandas: 数据分析

```
