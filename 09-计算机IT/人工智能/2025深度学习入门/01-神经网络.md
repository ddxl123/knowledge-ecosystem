### 一、神经元与感知机

```python
import numpy as np

# 单个神经元
class Neuron:
    def __init__(self, n_inputs):
        self.weights = np.random.randn(n_inputs) * 0.01
        self.bias = 0
    
    def forward(self, x):
        z = np.dot(self.weights, x) + self.bias
        return sigmoid(z)

# 感知机（二分类）
def perceptron(x, w, b):
    z = np.dot(w, x) + b
    return 1 if z >= 0 else 0

# 多层感知机（MLP）
# 输入层 → 隐藏层 → 输出层
# 每层包含多个神经元，层间全连接
```

### 二、激活函数

```python
# Sigmoid：输出 (0, 1)，适合二分类输出层
def sigmoid(z):
    return 1 / (1 + np.exp(-z))

# Tanh：输出 (-1, 1)，零中心化
def tanh(z):
    return np.tanh(z)

# ReLU：max(0, z)，最常用的隐藏层激活函数
def relu(z):
    return np.maximum(0, z)

# Leaky ReLU：解决 ReLU 死亡问题
def leaky_relu(z, alpha=0.01):
    return np.where(z > 0, z, alpha * z)

# Softmax：多分类输出层，输出概率分布
def softmax(z):
    exp_z = np.exp(z - np.max(z))  # 数值稳定性
    return exp_z / exp_z.sum()
```

- **激活函数选择：**
  - 隐藏层：ReLU（默认）或 Leaky ReLU
  - 二分类输出：Sigmoid
  - 多分类输出：Softmax
  - 回归输出：线性（无激活）

### 三、损失函数

```python
# 均方误差（MSE，回归）
def mse_loss(y_pred, y_true):
    return np.mean((y_pred - y_true) ** 2)

# 交叉熵（分类）
def cross_entropy(y_pred, y_true):
    return -np.sum(y_true * np.log(y_pred + 1e-8))

# 二元交叉熵
def binary_cross_entropy(y_pred, y_true):
    return -np.mean(y_true * np.log(y_pred + 1e-8) + 
                    (1 - y_true) * np.log(1 - y_pred + 1e-8))
```

### 四、反向传播

```python
# 反向传播核心：链式法则
# ∂L/∂w = ∂L/∂z * ∂z/∂w

# 简单反向传播示例
class SimpleNN:
    def __init__(self):
        self.w1 = np.random.randn(3, 4) * 0.01  # 输入→隐藏
        self.b1 = np.zeros(4)
        self.w2 = np.random.randn(4, 1) * 0.01  # 隐藏→输出
        self.b2 = np.zeros(1)
    
    def forward(self, x):
        self.z1 = x @ self.w1 + self.b1
        self.a1 = relu(self.z1)
        self.z2 = self.a1 @ self.w2 + self.b2
        self.a2 = sigmoid(self.z2)
        return self.a2
    
    def backward(self, x, y, lr=0.01):
        m = x.shape[0]
        # 输出层梯度
        dz2 = self.a2 - y
        dw2 = self.a1.T @ dz2 / m
        db2 = np.sum(dz2, axis=0) / m
        # 隐藏层梯度
        dz1 = (dz2 @ self.w2.T) * (self.z1 > 0)  # ReLU 导数
        dw1 = x.T @ dz1 / m
        db1 = np.sum(dz1, axis=0) / m
        # 更新参数
        self.w2 -= lr * dw2
        self.b2 -= lr * db2
        self.w1 -= lr * dw1
        self.b1 -= lr * db1
```

### 五、优化器

```python
# SGD（随机梯度下降）
# w = w - lr * gradient
# 简单但收敛慢，可能陷入局部最优

# Momentum（动量）
# v = momentum * v - lr * gradient
# w = w + v
# 加速收敛，减少震荡

# Adam（自适应学习率）
class Adam:
    def __init__(self, lr=0.001, beta1=0.9, beta2=0.999, epsilon=1e-8):
        self.lr = lr
        self.beta1, self.beta2, self.epsilon = beta1, beta2, epsilon
        self.m, self.v, self.t = {}, {}, 0
    
    def update(self, params, grads):
        self.t += 1
        for key in params:
            if key not in self.m:
                self.m[key] = np.zeros_like(params[key])
                self.v[key] = np.zeros_like(params[key])
            self.m[key] = self.beta1 * self.m[key] + (1 - self.beta1) * grads[key]
            self.v[key] = self.beta2 * self.v[key] + (1 - self.beta2) * grads[key]**2
            m_hat = self.m[key] / (1 - self.beta1**self.t)
            v_hat = self.v[key] / (1 - self.beta2**self.t)
            params[key] -= self.lr * m_hat / (np.sqrt(v_hat) + self.epsilon)
```

- **优化器选择：**
  - SGD + Momentum：经典方案，泛化性好
  - Adam：收敛快，调参简单，适合大多数场景
  - AdamW：Adam + 权重衰减，推荐用于 Transformer
